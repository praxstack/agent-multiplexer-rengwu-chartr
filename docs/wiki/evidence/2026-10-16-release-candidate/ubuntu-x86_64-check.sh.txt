#!/bin/bash
set -euo pipefail
export DEBIAN_FRONTEND=noninteractive
apt-get update -qq
apt-get install -y --no-install-recommends python3 xvfb xauth mesa-vulkan-drivers fonts-dejavu-core /packages/*.deb
cd /packages
sha256sum -c SHA256SUMS-x86_64
mkdir -p /work/extracted
tar -xzf chartr-0.3.1-linux-x86_64.tar.gz -C /work/extracted
python3 /work/startup.py
