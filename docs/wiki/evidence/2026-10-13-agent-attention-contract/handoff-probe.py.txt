transcript=[]
def record(label,data):
    transcript.append({'label':label,'data':data})
    print(json.dumps(transcript[-1]),flush=True)
def conn():
    s=socket.socket(socket.AF_UNIX);s.settimeout(5);s.connect(str(sockpath));return s
def request(method,params={}):
    with conn() as s:
        s.sendall((json.dumps({'id':'probe','method':method,'params':params})+'\n').encode())
        result=json.loads(s.makefile('rb').readline());record(method,result);return result
class Stream:
    def __init__(self,subscriptions):
        self.sock=conn();self.buf=b''
        self.sock.sendall((json.dumps({'id':'sub','method':'events.subscribe','params':{'subscriptions':subscriptions}})+'\n').encode())
    def drain(self,seconds=.4):
        until=time.monotonic()+seconds;lines=[]
        while time.monotonic()<until:
            if not select.select([self.sock],[],[],max(0,until-time.monotonic()))[0]:break
            part=self.sock.recv(65536)
            if not part:break
            self.buf+=part
            while b'\n' in self.buf:
                line,self.buf=self.buf.split(b'\n',1)
                if line:lines.append(json.loads(line))
        return lines
    def close(self):self.sock.close()

    record('identity',{'pid':server.pid,'root':str(root),'binary':str(BIN),'sha256':hashlib.sha256(BIN.read_bytes()).hexdigest()})
    deadline=time.monotonic()+10
    while not sockpath.exists() and server.poll() is None and time.monotonic()<deadline:time.sleep(.05)

    assert before['pane_id']==after['pane_id']
    assert before['terminal_id']!=after['terminal_id'],'source expectation changed: inspect identity preservation'
    record('old_subscription_after_handoff',stream.drain(.5))

finally:

    deadline=time.monotonic()+5
