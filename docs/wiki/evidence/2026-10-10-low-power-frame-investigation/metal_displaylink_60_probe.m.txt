static void draw_ready(id<CAMetalDrawable> supplied){

  double t=now();id<CAMetalDrawable> drawable=supplied;double ready=now();

  pass.colorAttachments[0].texture=drawable.texture;pass.colorAttachments[0].loadAction=MTLLoadActionClear;pass.colorAttachments[0].storeAction=MTLStoreActionStore;

  unsigned frame_id=frames+1;
  [command addCompletedHandler:^(id<MTLCommandBuffer> completed){fprintf(out,"{\"type\":\"gpu\",\"n\":%u,\"gpu_ms\":%.6f}\n",frame_id,(completed.GPUEndTime-completed.GPUStartTime)*1000);}];
  [drawable addPresentedHandler:^(id<MTLDrawable> shown){fprintf(out,"{\"type\":\"presented\",\"n\":%u,\"presented_s\":%.9f}\n",frame_id,shown.presentedTime);}];
  [command presentDrawable:drawable];[command commit];frames++;
  fprintf(out,"{\"type\":\"frame\",\"n\":%u,\"t_ms\":%.6f,\"wait_ms\":%.6f,\"encode_commit_ms\":%.6f}\n",frames,(t-start)*1000,(ready-t)*1000,(now()-ready)*1000);fflush(out);
 }

}
