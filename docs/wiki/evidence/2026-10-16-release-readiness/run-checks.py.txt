import json
import os
import pathlib
import subprocess
import sys
import time

root = pathlib.Path(__file__).resolve().parents[4]
evidence = pathlib.Path(__file__).resolve().parent
groups = {
    'rust': [
        ('workspace', ['cargo', 'test', '--workspace', '--locked', '--no-fail-fast']),
        ('live-session', ['cargo', 'test', '-p', 'chartr', '--test', 'live_session', '--locked', '--', '--ignored', '--nocapture', '--test-threads=1']),
        ('native-plugin', ['cargo', 'check', '--manifest-path', 'examples/plugins/hello/Cargo.toml', '--locked']),
        ('normal-build', ['cargo', 'build', '-p', 'chartr', '--locked']),
    ],
    'fast': [
        ('format', ['cargo', 'fmt', '--all', '--check']),
        ('scripts', ['python3', '-B', '-m', 'unittest', 'discover', '-s', 'scripts/tests', '-v']),
        ('javascript', ['node', '--test', 'crates/chartr/tests/plugin_bridge.cjs', 'plugins/wayfinder/tests/layout.test.mjs']),
    ],
    'metal': [
        ('metal-native', ['sh', 'scripts/check-metal-display-link.sh']),
    ],
    'backend': [
        ('backend-handoff', ['cargo', '+1.96.0', 'test', '--locked', '--manifest-path',
         'vendor/herdr/.build/source/5a2dee700eeeea68267a4d16777307632f77172f-9ed02822c8773c63053cf4d42745fff8758ef5aa57ddb5d4c3c55eec56bdaf70/Cargo.toml',
         'handoff_preserves_all_terminal_identities_and_invariants']),
    ],
}
group = sys.argv[1]
environment = os.environ.copy()
if group == 'backend':
    environment['CARGO_TARGET_DIR'] = str(root / 'vendor/herdr/.build/target')
    environment['ZIG'] = '/opt/homebrew/opt/zig@0.15/bin/zig'
results = []
for name, command in groups[group]:
    print(f'Starting {name}', flush=True)
    start = time.monotonic()
    log = evidence / f'{name}.log'
    with log.open('w') as output:
        result = subprocess.run(command, cwd=root, env=environment, stdout=output, stderr=subprocess.STDOUT)
    record = {'name': name, 'command': command, 'exit_code': result.returncode, 'seconds': round(time.monotonic()-start, 1)}
    results.append(record)
    (evidence / f'{group}-results.json').write_text(json.dumps(results, indent=2) + '\n')
    print(json.dumps(record), flush=True)
    if result.returncode:
        print('\n'.join(log.read_text().splitlines()[-30:]), flush=True)
print(f'Completed {group}', flush=True)
